rag_query.py 
import json
import sys
import numpy as np
from sentence_transformers import SentenceTransformer
from mlx_lm import load, generate

embedder = SentenceTransformer("all-MiniLM-L6-v2")
embeddings = np.load("rag_embeddings.npy")
chunks = json.load(open("rag_chunks.json"))

model, tokenizer = load("fused_llama_8bit")


def retrieve(query, k=3):
    q_emb = embedder.encode([query], normalize_embeddings=True)[0]
    scores = embeddings @ q_emb
    top_idx = np.argsort(scores)[::-1][:k]
    return [chunks[i] for i in top_idx]


def answer(query):
    context = "\n".join(retrieve(query))
    prompt = (
        "Answer the question using only the context below. "
        "If the answer isn't in the context, say you don't know.\n\n"
        f"Context:\n{context}\n\nQuestion: {query}\nAnswer:"
    )
    return generate(model, tokenizer, prompt=prompt, max_tokens=256)


if __name__ == "__main__":
    query = " ".join(sys.argv[1:]) or "where is seeroo it solutions located"
    print(answer(query))

rag_query_web.py 
import json
import sys
import numpy as np
from sentence_transformers import SentenceTransformer
from ddgs import DDGS
from mlx_lm import load, generate

embedder = SentenceTransformer("all-MiniLM-L6-v2")
embeddings = np.load("rag_embeddings.npy")
chunks = json.load(open("rag_chunks.json"))

model, tokenizer = load("fused_llama_8bit")


def retrieve_local(query, k=3):
    q_emb = embedder.encode([query], normalize_embeddings=True)[0]
    scores = embeddings @ q_emb
    top_idx = np.argsort(scores)[::-1][:k]
    return [chunks[i] for i in top_idx]


def retrieve_web(query, k=3):
    try:
        results = DDGS().text(query, max_results=k)
    except Exception as e:
        print(f"[web search failed: {e}]", file=sys.stderr)
        return []
    return [f"{r['title']}: {r['body']}" for r in results]


def answer(query, use_web=True):
    local_chunks = retrieve_local(query)
    web_chunks = retrieve_web(query) if use_web else []

    context_parts = []
    if local_chunks:
        context_parts.append("From local knowledge base:\n" + "\n".join(local_chunks))
    if web_chunks:
        context_parts.append("From web search:\n" + "\n".join(web_chunks))
    context = "\n\n".join(context_parts)

    prompt = (
        "Answer the question using only the context below. "
        "If the answer isn't in the context, say you don't know.\n\n"
        f"Context:\n{context}\n\nQuestion: {query}\nAnswer:"
    )
    return generate(model, tokenizer, prompt=prompt, max_tokens=256)


if __name__ == "__main__":
    query = " ".join(sys.argv[1:]) or "where is seeroo it solutions located"
    print(answer(query))

summarize.py 
import argparse
import os
import sys
from mlx_lm import load, generate

DEFAULT_MODEL = "mlx-community/Llama-3.2-3B-Instruct-4bit"
CHUNK_WORDS = 1200


def extract_text(path):
    ext = os.path.splitext(path)[1].lower()

    if ext == ".pdf":
        from pypdf import PdfReader
        reader = PdfReader(path)
        return "\n".join(page.extract_text() or "" for page in reader.pages)

    if ext == ".docx":
        from docx import Document
        doc = Document(path)
        return "\n".join(p.text for p in doc.paragraphs)

    if ext == ".txt":
        with open(path, encoding="utf-8") as f:
            return f.read()

    raise ValueError(f"Unsupported file type: {ext} (use .pdf, .docx, or .txt)")


def chunk_text(text, chunk_words=CHUNK_WORDS):
    words = text.split()
    return [
        " ".join(words[i:i + chunk_words])
        for i in range(0, len(words), chunk_words)
    ]


def summarize_chunk(model, tokenizer, text, max_tokens):
    prompt = (
        "Summarize the following text concisely, capturing the key points:\n\n"
        f"{text}\n\nSummary:"
    )
    return generate(model, tokenizer, prompt=prompt, max_tokens=max_tokens)


def summarize_document(model, tokenizer, text, max_tokens=300):
    chunks = chunk_text(text)
    print(f"[{len(chunks)} chunk(s), {len(text.split())} words total]", file=sys.stderr)

    if len(chunks) == 1:
        return summarize_chunk(model, tokenizer, chunks[0], max_tokens)

    chunk_summaries = []
    for i, chunk in enumerate(chunks, 1):
        print(f"[summarizing chunk {i}/{len(chunks)}]", file=sys.stderr)
        chunk_summaries.append(summarize_chunk(model, tokenizer, chunk, max_tokens))

    combined = "\n\n".join(chunk_summaries)
    final_prompt = (
        "Combine the following section summaries into one coherent overall summary:\n\n"
        f"{combined}\n\nOverall summary:"
    )
    return generate(model, tokenizer, prompt=final_prompt, max_tokens=max_tokens)


def main():
    parser = argparse.ArgumentParser(description="Summarize a PDF, DOCX, or TXT file with MLX/Llama.")
    parser.add_argument("file", help="Path to a .pdf, .docx, or .txt file")
    parser.add_argument("--model", default=DEFAULT_MODEL, help="MLX model path or HF repo")
    parser.add_argument("--adapter-path", default=None, help="Optional LoRA adapter path")
    parser.add_argument("--max-tokens", type=int, default=300, help="Max tokens per summary")
    args = parser.parse_args()

    text = extract_text(args.file)
    if not text.strip():
        print("No extractable text found in file.", file=sys.stderr)
        sys.exit(1)

    load_kwargs = {"adapter_path": args.adapter_path} if args.adapter_path else {}
    model, tokenizer = load(args.model, **load_kwargs)

    summary = summarize_document(model, tokenizer, text, args.max_tokens)
    print(summary)


if __name__ == "__main__":
    main()
